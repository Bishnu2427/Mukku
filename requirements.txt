# =============================================================================
#  Mukku AI Studio — Python dependencies
# =============================================================================
#  Pinned to exact versions (OWASP A06: Vulnerable and Outdated Components).
#  Unpinned ranges mean two builds of the same commit can install different
#  code, so a compromised or broken release lands silently and the image you
#  tested is not the image you shipped. These are the versions verified working
#  in the production container on 2026-08-23.
#
#  Upgrading:
#    pip install pip-audit && pip-audit -r requirements.txt   # check CVEs
#    pip install -U <pkg> && pip freeze | grep <pkg>          # re-pin
#    docker build -t mukku:latest .                           # re-verify
# =============================================================================

# ── Web framework ────────────────────────────────────────────────────────────
flask==3.1.3
# click is left unpinned: gTTS 2.5.4 (the newest release) caps it at <8.2, so
# the CVE-fixed click 8.3.3 cannot coexist with it. See the accepted-risk note
# for PYSEC-2026-2132 in SECURITY.md — replacing gTTS with Piper (roadmap T2-2)
# removes the constraint entirely.
flask-cors==6.0.5
python-dotenv==1.2.3

# ── Auth ─────────────────────────────────────────────────────────────────────
bcrypt==5.0.0
flask-limiter==4.1.1

# Google OAuth uses the Python stdlib (urllib) — no extra library needed.

# ── Database ─────────────────────────────────────────────────────────────────

# ── Job queue + shared rate-limit storage ────────────────────────────────────

redis==8.1.0
rq==2.11.0

# ── Production WSGI server ───────────────────────────────────────────────────

# ── Imaging ──────────────────────────────────────────────────────────────────

# ── Local Stable Diffusion (OPTIONAL) ────────────────────────────────────────
# Moved to requirements-optional-sd.txt — SKIP_SD=true means it never runs, it

# you actually want local SD on a CUDA box.

# ── Voice ────────────────────────────────────────────────────────────────────
# pyttsx3 drives Windows SAPI; on Linux set TTS_ENGINE=gtts.
# Coqui TTS is excluded — it does not support Python 3.12+.
pyttsx3==2.99
gTTS==2.5.4

# ── Video assembly ───────────────────────────────────────────────────────────
# moviepy 2.x: imports from `moviepy` directly, not `moviepy.editor`.
#
# Installed with --no-deps (see below) because moviepy 2.2.1 declares
# pillow<12.0, which would hold Pillow on 11.3.0 — a release with five known
# image-parsing CVEs. Mukku decodes user-uploaded images with Pillow, so those
# are live attack surface, not theoretical. moviepy was verified working on
# Pillow 12.3.0 (import, ColorClip, Ken Burns render, full pipeline e2e), so
# the cap is conservative metadata rather than a real incompatibility.
#
# Its real runtime deps are listed here explicitly so nothing is lost:
decorator==4.4.2
proglog==0.1.12
imageio==2.37.2
imageio-ffmpeg==0.6.0

# ── AI providers ─────────────────────────────────────────────────────────────
google-genai==2.19.0
